% Name: Weighted graph
% Description: Distances between five towns, written on the edges.
% Tags: weighted graph, distances, edges

\begin{tikzpicture}[
  town/.style={circle, draw, thick, minimum size=9mm, font=\small},
  weight/.style={font=\footnotesize, inner sep=1.5pt},
]
  \node[town] (A) at (0,0) {A};
  \node[town] (B) at (3,1.6) {B};
  \node[town] (C) at (3,-1.6) {C};
  \node[town] (D) at (6,1.2) {D};
  \node[town] (E) at (6,-1.2) {E};

  \foreach \u/\v/\w in {A/B/12, A/C/9, B/C/4, B/D/7, C/E/11, D/E/3, C/D/8}
    \draw[thick, gray!70!black] (\u) -- node[weight] {\w} (\v);
\end{tikzpicture}
