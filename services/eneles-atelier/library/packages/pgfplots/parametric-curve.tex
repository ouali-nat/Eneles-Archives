% Name: Parametric curve
% Description: A Lissajous figure, x = sin 3t and y = sin 4t, on equal axes.
% Tags: parametric, Lissajous, curve

\usepackage{pgfplots}
\pgfplotsset{compat=1.18}

\begin{tikzpicture}
  \begin{axis}[
    width=7cm, height=7cm, axis equal,
    axis lines=middle, xlabel=$x$, ylabel=$y$,
    xmin=-1.3, xmax=1.3, ymin=-1.3, ymax=1.3,
    xtick={-1,1}, ytick={-1,1},
  ]
    \addplot[blue!70!black, thick, domain=0:360, samples=300, variable=\t]
      ({sin(3*t)}, {sin(4*t)});
  \end{axis}
\end{tikzpicture}
