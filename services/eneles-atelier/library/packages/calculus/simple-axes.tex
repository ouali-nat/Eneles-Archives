% Name: Axes and a function
% Description: Two arrows for the axes and one plotted function: the plainest graph to start from.
% Tags: axes, function, plot

\begin{tikzpicture}[domain=-2:2, samples=50]
  \draw[->] (-2.5,0) -- (2.5,0) node[right] {$x$};
  \draw[->] (0,-0.5) -- (0,4.5) node[above] {$y$};
  \draw[thick] plot (\x, {\x*\x}) node[right] {$y = x^2$};
\end{tikzpicture}
