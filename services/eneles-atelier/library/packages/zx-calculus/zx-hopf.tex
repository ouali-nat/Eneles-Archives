% Name: Hopf rule
% Description: A Z and an X spider joined by two wires come apart: the double connection disappears.
% Tags: Hopf rule, rewrite rule, ZX

\usepackage{zx-calculus}

$\begin{ZX}
  \zxN{} \rar & \zxZ{} \ar[r, bend left] \ar[r, bend right] & \zxX{} \rar & \zxN{}
\end{ZX}
=
\begin{ZX}
  \zxN{} \rar & \zxZ{} & \zxX{} \rar & \zxN{}
\end{ZX}$
